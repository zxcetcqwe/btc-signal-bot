608: def detect_candidate(df1, df5, df15, regime, levels, ob_imbalance):

610:     prev = df1.iloc[-2]
611: 
612:     close = float(last["close"])

618:     ema20 = float(last["ema20"])
619:     ema50 = float(last["ema50"])

621:     local_support = levels["local_support"]
622:     local_resistance = levels["local_resistance"]
623: 
624:     candidates = []
625: 
626:     breakout_buffer = 0.10 * atr_now
627:     sl_buffer = 0.35 * atr_now
628: 
629:     # 1) Breakout long: цена закрылась выше сопротивления.
630:     if prev_close <= local_resistance and close > local_resistance + breakout_buffer:
631:         entry = local_resistance + 0.05 * atr_now
632:         sl = min(local_resistance - sl_buffer, float(df1["low"].iloc[-8:].min()) - 0.10 * atr_now)
633:         tp = entry + (entry - sl) * MIN_RR
634:         candidates.append(make_candidate(
635:             "LONG",

638:             sl,
639:             tp,

641:         ))

643:     # 2) Breakdown short: цена закрылась ниже поддержки.
644:     if prev_close >= local_support and close < local_support - breakout_buffer:
645:         entry = local_support - 0.05 * atr_now
646:         sl = max(local_support + sl_buffer, float(df1["high"].iloc[-8:].max()) + 0.10 * atr_now)
647:         tp = entry - (sl - entry) * MIN_RR
648:         candidates.append(make_candidate(
649:             "SHORT",
650:             "breakdown_retest_short",
651:             entry,
652:             sl,
653:             tp,
654:             "Пробой локальной поддержки; план — вход на ретесте пробитой зоны снизу."
655:         ))
656: 
657:     # 3) Pullback long.
658:     if regime == "TREND_UP":
659:         touched_ema = low <= ema20 or low <= ema50
660:         reclaimed = close > ema20 and close > open_
661:         if touched_ema and reclaimed:
662:             entry = close
663:             sl = min(low, float(df1["low"].iloc[-8:].min())) - sl_buffer
664:             tp = entry + (entry - sl) * MIN_RR
665:             candidates.append(make_candidate(
666:                 "LONG",
667:                 "pullback_long",
668:                 entry,
669:                 sl,
670:                 tp,
671:                 "Откат к EMA в восходящем режиме и закрытие обратно выше EMA20."
672:             ))
673: 
674:     # 4) Pullback short.
675:     if regime == "TREND_DOWN":
676:         touched_ema = high >= ema20 or high >= ema50
677:         rejected = close < ema20 and close < open_
678:         if touched_ema and rejected:
679:             entry = close
680:             sl = max(high, float(df1["high"].iloc[-8:].max())) + sl_buffer
681:             tp = entry - (sl - entry) * MIN_RR
682:             candidates.append(make_candidate(
683:                 "SHORT",
684:                 "pullback_short",
685:                 entry,
686:                 sl,
687:                 tp,
688:                 "Откат к EMA в нисходящем режиме и закрытие обратно ниже EMA20."
689:             ))
690: 
691:     valid = []
692:     for c in candidates:
